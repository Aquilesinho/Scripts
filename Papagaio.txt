local Players = game:GetService("Players")
local TextChatService = game:GetService("TextChatService")
local UserInputService = game:GetService("UserInputService")

local LocalPlayer = Players.LocalPlayer

local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "ChatMirrorGUI"
ScreenGui.ResetOnSpawn = false
ScreenGui.Parent = LocalPlayer:WaitForChild("PlayerGui")

local Frame = Instance.new("Frame")
Frame.Size = UDim2.new(0, 300, 0, 450)
Frame.Position = UDim2.new(0, 50, 0, 100)
Frame.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
Frame.BorderSizePixel = 0
Frame.Parent = ScreenGui

local Title = Instance.new("TextLabel")
Title.Size = UDim2.new(1, 0, 0, 40)
Title.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
Title.TextColor3 = Color3.new(1, 1, 1)
Title.Text = "Escolha um jogador"
Title.TextSize = 18
Title.Parent = Frame

-- BOTÃO CHAT
local ChatButton = Instance.new("TextButton")
ChatButton.Position = UDim2.new(0, 10, 0, 50)
ChatButton.Size = UDim2.new(0, 135, 0, 35)
ChatButton.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
ChatButton.TextColor3 = Color3.new(1, 1, 1)
ChatButton.Text = "☐ Chat"
ChatButton.TextSize = 16
ChatButton.Parent = Frame

-- BOTÃO FOLLOW
local FollowButton = Instance.new("TextButton")
FollowButton.Position = UDim2.new(0, 155, 0, 50)
FollowButton.Size = UDim2.new(0, 135, 0, 35)
FollowButton.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
FollowButton.TextColor3 = Color3.new(1, 1, 1)
FollowButton.Text = "☐ Follow"
FollowButton.TextSize = 16
FollowButton.Parent = Frame

-- TODOS
local AllButton = Instance.new("TextButton")
AllButton.Position = UDim2.new(0, 10, 0, 95)
AllButton.Size = UDim2.new(1, -20, 0, 35)
AllButton.BackgroundColor3 = Color3.fromRGB(80, 60, 140)
AllButton.TextColor3 = Color3.new(1, 1, 1)
AllButton.Text = "TODOS"
AllButton.TextSize = 16
AllButton.Parent = Frame

-- LISTA
local List = Instance.new("ScrollingFrame")
List.Position = UDim2.new(0, 10, 0, 140)
List.Size = UDim2.new(1, -20, 0, 300)
List.BackgroundTransparency = 1
List.BorderSizePixel = 0
List.CanvasSize = UDim2.new(0, 0, 0, 0)
List.Parent = Frame

local Layout = Instance.new("UIListLayout")
Layout.Padding = UDim.new(0, 5)
Layout.Parent = List

local selectedPlayer = nil
local allPlayers = false

local chatEnabled = false
local followEnabled = false
local enviando = false

-- Atualiza aparência dos botões
local function updateModeButtons()
	if chatEnabled then
		ChatButton.Text = "☑ Chat"
		ChatButton.BackgroundColor3 = Color3.fromRGB(70, 120, 200)
	else
		ChatButton.Text = "☐ Chat"
		ChatButton.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
	end

	if followEnabled then
		FollowButton.Text = "☑ Follow"
		FollowButton.BackgroundColor3 = Color3.fromRGB(70, 180, 100)
	else
		FollowButton.Text = "☐ Follow"
		FollowButton.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
	end
end

local function updateCanvas()
	List.CanvasSize = UDim2.new(
		0,
		0,
		0,
		Layout.AbsoluteContentSize.Y + 10
	)
end

-- CHAT
ChatButton.MouseButton1Click:Connect(function()
	chatEnabled = not chatEnabled
	updateModeButtons()
end)

-- FOLLOW
FollowButton.MouseButton1Click:Connect(function()
	if allPlayers then
		-- Follow não funciona com TODOS
		followEnabled = false
		updateModeButtons()
		return
	end

	followEnabled = not followEnabled
	updateModeButtons()
end)

-- TODOS
AllButton.MouseButton1Click:Connect(function()
	allPlayers = true
	selectedPlayer = nil

	Title.Text = "Seguindo: TODOS"

	-- Follow não pode funcionar em TODOS
	followEnabled = false

	AllButton.BackgroundColor3 = Color3.fromRGB(100, 180, 100)

	for _, button in ipairs(List:GetChildren()) do
		if button:IsA("TextButton") then
			button.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
		end
	end

	updateModeButtons()
end)

-- CRIA BOTÃO DE JOGADOR
local function createPlayerButton(player)
	local Button = Instance.new("TextButton")

	Button.Size = UDim2.new(1, 0, 0, 35)
	Button.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
	Button.TextColor3 = Color3.new(1, 1, 1)

	Button.Text =
		player.DisplayName ..
		" (@" ..
		player.Name ..
		")"

	Button.TextSize = 16
	Button.Parent = List

	Button.MouseButton1Click:Connect(function()
		selectedPlayer = player
		allPlayers = false

		Title.Text = "Alvo: " .. player.DisplayName

		AllButton.BackgroundColor3 =
			Color3.fromRGB(80, 60, 140)

		for _, otherButton in ipairs(List:GetChildren()) do
			if otherButton:IsA("TextButton") then
				otherButton.BackgroundColor3 =
					Color3.fromRGB(50, 50, 50)
			end
		end

		Button.BackgroundColor3 =
			Color3.fromRGB(70, 120, 200)
	end)
end

-- REFAZ LISTA
local function rebuildPlayers()
	for _, child in ipairs(List:GetChildren()) do
		if child:IsA("TextButton") then
			child:Destroy()
		end
	end

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer then
			createPlayerButton(player)
		end
	end

	task.wait()
	updateCanvas()
end

Players.PlayerAdded:Connect(function()
	rebuildPlayers()
end)

Players.PlayerRemoving:Connect(function(player)
	if player == selectedPlayer then
		selectedPlayer = nil
		followEnabled = false
		Title.Text = "Escolha um jogador"
		updateModeButtons()
	end

	rebuildPlayers()
end)

-- ENVIA CHAT
local function enviarMensagem(texto)
	if enviando then
		return
	end

	if texto == "" then
		return
	end

	enviando = true

	task.wait(0.05)

	local chat = TextChatService.TextChannels:FindFirstChild("RBXGeneral")

	if chat then
		chat:SendAsync(texto)
	end

	task.wait(0.2)

	enviando = false
end

-- RECEBE MENSAGENS
TextChatService.MessageReceived:Connect(function(message)
	if not chatEnabled then
		return
	end

	if not message.TextSource then
		return
	end

	local sender =
		Players:GetPlayerByUserId(message.TextSource.UserId)

	if not sender then
		return
	end

	if sender == LocalPlayer then
		return
	end

	if allPlayers then
		enviarMensagem(message.Text)
		return
	end

	if selectedPlayer and sender == selectedPlayer then
		enviarMensagem(message.Text)
	end
end)

-- FOLLOW
task.spawn(function()
	while true do
		task.wait(0.2)

		if followEnabled
			and not allPlayers
			and selectedPlayer
			and selectedPlayer.Character
			and LocalPlayer.Character then

			local targetRoot =
				selectedPlayer.Character:FindFirstChild("HumanoidRootPart")

			local myRoot =
				LocalPlayer.Character:FindFirstChild("HumanoidRootPart")

			if targetRoot and myRoot then
				myRoot.CFrame =
					targetRoot.CFrame + Vector3.new(0, 3, 0)
			end
		end
	end
end)

-- ARRASTAR GUI
local dragging = false
local dragStart
local startPosition

Title.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 then
		dragging = true
		dragStart = input.Position
		startPosition = Frame.Position

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

Title.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement then
		local connection

		connection = UserInputService.InputChanged:Connect(function(changedInput)
			if changedInput == input and dragging then
				local delta =
					changedInput.Position - dragStart

				Frame.Position = UDim2.new(
					startPosition.X.Scale,
					startPosition.X.Offset + delta.X,

					startPosition.Y.Scale,
					startPosition.Y.Offset + delta.Y
				)
			end
		end)

		task.delay(0.1, function()
			if connection then
				connection:Disconnect()
			end
		end)
	end
end)

rebuildPlayers()
updateModeButtons()
